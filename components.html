Add dynamic, interactive particle backgrounds to any element or the whole page.

Interactive: Bursting Bubbles (Hover to Pop)

new ParticleEngine('canvasBurst', {
    particleColor: 'rgba(96, 165, 250, 0.6)',
    lineColor: '0, 0, 0', // No lines
    particleCount: 40,
    maxDistance: 0,
    sizeMultiplier: 6,
    speedMultiplier: 0.5,
    interactiveMode: 'burst',
    interactionRadius: 80
});

Interactive: Repulsive Network (Hover to Push)

new ParticleEngine('canvasRepulse', {
    particleColor: 'rgba(239, 68, 68, 0.8)',
    lineColor: '239, 68, 68',
    particleCount: 120,
    maxDistance: 100,
    speedMultiplier: 1.5,
    interactiveMode: 'repulse',
    interactionRadius: 100
});
